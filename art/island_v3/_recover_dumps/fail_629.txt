def _extrude_shards(bm: bmesh.types.BMesh) -> None:
    """Push a few side faces out as fused angular blocks, still one mesh."""
    bm.faces.ensure_lookup_table()
    ranked: list[tuple[float, bmesh.types.BMFace]] = []
    for f in bm.faces:
        if not f.is_valid:
            continue
        f.normal_update()
        n = f.normal
        c = f.calc_center_median()
        if c.z > -6.0 or c.z < -198.0:
            continue
        if n.z > 0.58 or abs(n.z) > 0.86:
            continue
        area = f.calc_area()
        if area < 400.0 or area > 24000.0:
            continue
        if _u01(int(c.x), int(c.y), 41) < 0.34:
            continue
        ranked.append((area, f))
    ranked.sort(key=lambda it: it[0], reverse=True)
    for i, (_, f) in enumerate(ranked[:16]):
        if not f.is_valid:
            continue
        f.normal_update()
        n = f.normal.copy()
        c = f.calc_center_median()
        radial = Vector((c.x, c.y, 0.12 * c.z))
        if radial.length > 1e-6:
            radial.normalize()
            n = (0.50 * n + 0.50 * radial).normalized()
        dist = 28.0 + 44.0 * _u01(i, 41, 7)
        geom = bmesh.ops.extrude_face_region(bm, geom=[f])
        verts = [e for e in geom["geom"] if isinstance(e, bmesh.types.BMVert)]
        faces = [e for e in geom["geom"] if isinstance(e, bmesh.types.BMFace)]
        if not verts:
            continue
        bmesh.ops.translate(bm, verts=verts, vec=n * dist)
        cent = Vector((0.0, 0.0, 0.0))
        for v in verts:
